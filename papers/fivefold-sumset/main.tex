\begin{abstract}
We give finite sets of integers $A,B$ satisfying
$|A-B|^5>|A+B|^5|5B|$, where $5B$ is the fivefold sumset of $B$.
The sets are base-$193$ encodings of nonnegative integer vectors in
dimension $128$, with coordinate sums bounded by $160$ and $32$,
respectively. Stars and bars counts the sumsets, and a sum of $33$
binomial terms counts the difference set. The strict comparison uses
integer arithmetic and is independently reproduced by counting both
positive and negative supports.
A second construction, using a fixed-weight binary digit set, gives a
finite proof by type counting and positive generating polynomials; its
strict comparison follows from small rational inequalities and the
binomial theorem.
\end{abstract}
\medskip\noindent\textbf{Keywords.} sumset; difference set; integer simplex; type counting; exact certificate.
\smallskip\noindent\textbf{MSC 2020.} 11B75; 05A16.

\section{The inequality and the construction}
For finite nonempty subsets $A,B$ of an abelian group, write
$A\pm B=\{a\pm b:a\in A,\ b\in B\}$ and
$hB=\{b_1+\cdots+b_h:b_1,\ldots,b_h\in B\}$.
Gyarmati, Hennecart and Ruzsa~\cite[equation~(9), p.~177]{ghr} asked for which
integers $h$ the inequality
\begin{equation}\label{eq:historical}
 |A-B|\le |A+B|\,|hB|^{1/h}
\end{equation}
holds for all such sets. Their discussion on p.~182 gives a counterexample
for $h=6$ and leaves $h=4,5$ unresolved. The result below gives an
explicit counterexample to the $h=5$ statement in that text; no priority
or current-status claim is made.

For integers $m\ge1$ and $r\ge0$, define the integer simplex
\[
 \V(m,r)=\left\{x\in\Z_{\ge0}^{m}:
                  \sum_{i=0}^{m-1}x_i\le r\right\}.
\]
Its size is $\binom{m+r}{m}$. The lattice-simplex construction and its
equal-radius difference count appear in Hennecart, Robert and
Yudin~\cite[Section~3]{hry}; Gyarmati, Hennecart and Ruzsa use this family
for their $h=6$ example~\cite[p.~182]{ghr}.
Glasscock~\cite[Lemma~B.1, pp.~37--38]{glasscock} gives the exact count
for unequal radii, including the negative-support proof reproduced below.
His Question~8 on p.~28 also leaves $h=4,5$ unresolved.
Our example specializes that known formula to $(m,a,b)=(128,160,32)$
and verifies the strict fivefold comparison.

\begin{samepage}
\begin{theorem}\label{thm:main}
Let $\phi:\Z^{128}\to\Z$ be the additive map
\[
 \phi(x)=\sum_{i=0}^{127}193^i x_i.
\]
The finite integer sets
\[
 A=\phi\bigl(\V(128,160)\bigr),\qquad
 B=\phi\bigl(\V(128,32)\bigr)
\]
satisfy
\begin{equation}\label{eq:failure}
 |A-B|^5>|A+B|^5|5B|.
\end{equation}
\end{theorem}
\end{samepage}

\section{Counting the vector sets}
We recall the unequal-radius count from~\cite[Lemma~B.1]{glasscock}
and give a second expression for its arithmetic verification.

\begin{lemma}\label{lem:counts}
For integers $m\ge1$ and $a,b\ge0$, let $X=\V(m,a)$ and $Y=\V(m,b)$. Then
\begin{align}
 |X+Y|&=\binom{m+a+b}{m}, &
 |5Y|&=\binom{m+5b}{m},\label{eq:sumcounts}\\
 |X-Y|&=\sum_{j=0}^{\min(m,b)}
       \binom mj\binom bj\binom{m-j+a}{m-j}.\label{eq:diffcount}
\end{align}
The difference count also equals
\begin{equation}\label{eq:double}
 \sum_{j=0}^{\min(m,b)}\ \sum_{k=0}^{\min(m-j,a)}
 \binom mj\binom{m-j}{k}\binom bj\binom ak.
\end{equation}
\end{lemma}

\begin{proof}
One has $\V(m,a)+\V(m,b)=\V(m,a+b)$. To obtain the reverse
inclusion, allocate up to $a$ units of any vector's coordinate mass
to the first summand; the remaining mass is at most $b$. Iterating
this identity gives $5Y=\V(m,5b)$. Stars and bars proves
\eqref{eq:sumcounts}.

For $z\in\Z^m$, put $(z_+)_i=\max(z_i,0)$ and
$(z_-)_i=\max(-z_i,0)$. Then
\[
 z\in X-Y\quad\Longleftrightarrow\quad
 \sum_i(z_+)_i\le a\ \text{ and }\ \sum_i(z_-)_i\le b.
\]
Necessity follows from any representation $z=x-y$; sufficiency follows
by taking $x=z_+$ and $y=z_-$. If $z$ has exactly $j$ negative
coordinates, their positions can be chosen in $\binom mj$ ways.
Their positive magnitudes have total at most $b$, giving $\binom bj$
choices. The other $m-j$ coordinates are nonnegative with total at
most $a$, giving $\binom{m-j+a}{m-j}$ choices. This proves
\eqref{eq:diffcount}, including the case $j=0$.

Alternatively, choose the $j$ negative positions and then the $k$
positive positions. Their nonzero magnitudes can be filled in
$\binom bj$ and $\binom ak$ ways. Summing over the disjoint supports
proves~\eqref{eq:double}.
\end{proof}

For the parameters of Theorem~\ref{thm:main}, set
\begin{align}
 D&=\sum_{j=0}^{32}
      \binom{128}{j}\binom{32}{j}\binom{288-j}{128-j},\label{eq:D}\\
 S&=\binom{320}{128},\qquad
 F=\binom{288}{128}.\label{eq:SF}
\end{align}
\begin{samepage}
These integers satisfy $D^5>S^5F$. The arithmetic can be checked with
the following standalone Python code, using integer operations only:
\begin{verbatim}
from math import comb
D = sum(comb(128,j)*comb(32,j)*comb(288-j,128-j)
        for j in range(33))
S, F = comb(320,128), comb(288,128)
if D**5 <= S**5*F:
    raise ArithmeticError("fivefold comparison failed")
print("D^5 > S^5 F")
\end{verbatim}
\end{samepage}
The accompanying verifier also evaluates~\eqref{eq:double} with
factorials instead of the binomial-coefficient routine and checks
the resulting integers against the stored values. Its small-set
controls enumerate the sets themselves.

\section{Encoding as integer sets}
\begin{proof}[Proof of Theorem~\ref{thm:main}]
Write $X=\V(128,160)$ and $Y=\V(128,32)$.
Every coordinate of $X+Y$ lies in $[0,192]$, and every coordinate
of $5Y$ lies in $[0,160]$. Base-$193$ uniqueness therefore makes
$\phi$ injective on both sets.

Every coordinate of $X-Y$ lies in $[-32,160]$. Adding $32$ to
each coordinate gives digits in $[0,192]$ and adds the same integer
$32\sum_{i=0}^{127}193^i$ to every image. Thus $\phi$ is injective
on $X-Y$ as well. Additivity now gives
\[
 |A-B|=D,\qquad |A+B|=S,\qquad |5B|=F.
\]
The integer comparison following~\eqref{eq:SF} proves
\eqref{eq:failure}.
\end{proof}

\section{A second construction by finite type counting}
\label{sec:binary-type}
The same fivefold reversal also has a construction whose strict comparison
uses only small rational inequalities and the binomial theorem. Its
dimension is larger than that of Theorem~\ref{thm:main}. The argument
specializes standard type counting, as used in the cardinality--entropy
passage of Lau and Nair~\cite[Theorem~1]{launair}, within the weighted
max-convolution framework of Matolcsi, Ruzsa, Shakan and
Zhelezov~\cite[Section~8]{mrsz}. We give the complete finite argument.

\begin{theorem}\label{thm:binary-type}
Set $k=2048$, $n=271k=555008$, and define
\[
 X=\left\{x\in\{0,\ldots,7\}^n:\sum_{i=0}^{n-1}x_i=247k\right\},
 \qquad
 Y=\left\{y\in\{0,1\}^n:\sum_{i=0}^{n-1}y_i=16k\right\}.
\]
For $\psi(v)=\sum_{i=0}^{n-1}9^iv_i$, the finite nonempty integer sets
$A=\psi(X)$ and $B=\psi(Y)$ satisfy $|A-B|^5>|A+B|^5|5B|$.
\end{theorem}

\begin{proof}
Consider all words with digit multiplicities
\[
 (m_{-1},m_0,m_1,\ldots,m_7)
       =k(16,128,64,32,16,8,4,2,1).
\]
Each belongs to $X-Y$: at a digit $-1$ take $(x_i,y_i)=(0,1)$;
at every other digit $d$ take $(x_i,y_i)=(d,0)$. The coordinate sums
are $16k$ for $y$ and
$(64+2\cdot32+3\cdot16+4\cdot8+5\cdot4+6\cdot2+7)k=247k$ for $x$.
These words are distinct difference vectors, so
\begin{equation}\label{eq:binary-type-lower}
 |X-Y|\ge \frac{n!}{\prod_{d=-1}^{7}m_d!}
       \ge \frac{Q^n2^{295k}}{(n+1)^9},
 \qquad Q=\frac{271}{128}.
\end{equation}
To see the second inequality directly, put $p_d=m_d/n$. In the
multinomial distribution with probabilities $p_d$, the displayed type
is modal: for each positive integer $m$, the sequence $m^j/j!$ is
maximal at $j=m$, and a type probability is a common factor times
$\prod_d m_d^{j_d}/j_d!$. There are at most $(n+1)^9$ types. Thus the
displayed type has probability at least $(n+1)^{-9}$. Substituting
$\prod_d p_d^{m_d}=Q^{-n}2^{-295k}$ proves the bound.

Every $X+Y$ output has digits in $\{0,\ldots,8\}$ and total $263k$.
Every $5Y$ output has digits in $\{0,\ldots,5\}$ and total $80k$.
For a polynomial with nonnegative coefficients, its coefficient of
$z^r$ is at most its value at any $z>0$, divided by $z^r$.
Apply this to $(1+z+\cdots+z^8)^n$ at $z=1/2$ and to
$(1+z+\cdots+z^5)^n$ at $z=1/4$. This gives
\begin{equation}\label{eq:binary-type-upper}
 |X+Y|\le P^n2^{263k},\qquad |5Y|\le T^n2^{160k},
 \qquad P=\frac{511}{256},\quad T=\frac{1365}{1024}.
\end{equation}
The coefficient bounds count all words with the prescribed digit totals;
they require no assertion that every such word is attainable.
Since $5\cdot295=5\cdot263+160$, the bounds imply
\[
 \frac{|X-Y|^5}{|X+Y|^5|5Y|}
 \ge\frac{R^n}{(n+1)^{45}},
 \qquad R=\frac{(542/511)^5}{1365/1024}.
\]
Small rational comparisons suffice:
\[
 \frac{542}{511}>\frac{53}{50},\qquad
 \left(\frac{53}{50}\right)^5>
 1+5\frac3{50}+10\left(\frac3{50}\right)^2=\frac{167}{125},
 \qquad
 \frac{171008}{170625}>\frac{513}{512}.
\]
The last fraction before $513/512$ is $(167/125)/(1365/1024)$;
its cross-product difference is $25471$. Hence $R>1+1/512$.
As $n=512\cdot1084$ and $n+1<2^{20}$, the binomial theorem yields
\[
 R^n>\bigl((1+1/512)^{512}\bigr)^{1084}
      >2^{1084}>2^{900}>(n+1)^{45}.
\]
This proves the strict vector-set inequality.

Finally, the digits of $X-Y$, $X+Y$ and $5Y$ lie in $[-1,7]$,
$[0,8]$ and $[0,5]$, respectively. Base $9$ is injective on each
of these vector sets: at a largest differing position $j$, the leading
contribution has magnitude at least $9^j$, while all lower contributions
have total magnitude at most $8\sum_{i<j}9^i=9^j-1$. Additivity of
$\psi$ therefore preserves all three cardinalities and proves the
integer-set assertion. In particular, $B$ consists of the
$\binom{n}{16k}$ base-$9$ words with exactly $16k$ digits equal to one.
\end{proof}
